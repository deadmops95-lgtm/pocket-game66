# --- БАЗА ДАННЫХ (SQLite) ---
def init_db():
    conn = sqlite3.connect("game.db")
    cursor = conn.cursor()
    # Таблица игроков
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS players (
            user_id INTEGER PRIMARY KEY,
            energy INTEGER DEFAULT 10,
            candies INTEGER DEFAULT 5,
            rating INTEGER DEFAULT 1200,
            wins INTEGER DEFAULT 0,
            loses INTEGER DEFAULT 0
        )
    """)
    # Таблица покемонов игрока
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS pokemons (
            user_id INTEGER PRIMARY KEY,
            name TEXT DEFAULT 'Флеймлинг',
            type TEXT DEFAULT 'Огонь',
            emoji TEXT DEFAULT '🔥',
            level INTEGER DEFAULT 1,
            exp INTEGER DEFAULT 0,
            max_exp INTEGER DEFAULT 50,
            hp INTEGER DEFAULT 100,
            max_hp INTEGER DEFAULT 100,
            atk INTEGER DEFAULT 15
        )
    """)
    conn.commit()
    conn.close()

# Функция регистрации нового игрока
def register_user(user_id):
    conn = sqlite3.connect("game.db")
    cursor = conn.cursor()
    cursor.execute("SELECT user_id FROM players WHERE user_id = ?", (user_id,))
    if not cursor.fetchone():
        cursor.execute("INSERT INTO players (user_id) VALUES (?)", (user_id,))
        cursor.execute("INSERT INTO pokemons (user_id) VALUES (?)", (user_id,))
        conn.commit()
    conn.close()
